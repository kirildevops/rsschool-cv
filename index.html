package kata

func RowSumOddNumbers(n int) int {

    oddNumbersToSkip := n * (n-1) / 2
    // ((first + last) / 2 ) * count; first = 1;
    // last = n - 1; count = n - 1;

    firstInTheRow := 2 * (oddNumbersToSkip + 1) - 1
    // 2 * x - 1 -- the formula for the first
    // in the row element; x = skippedCount + 1
    sum := (firstInTheRow + n - 1) * n
    // arithmetic progression. Again.
    // first = ...; last = f + 2 * (n - 1); count = n
    // ((first + last) / 2) * count = ((f + f + 2 * (n - 1)) / 2 ) * n = (f + n - 1) * n
    return sum
}
